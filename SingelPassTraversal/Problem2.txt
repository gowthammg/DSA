Given a binary array containing only 0 and 1, find the maximum number of consecutive ones.
Input: arr = [1, 1, 0, 1, 1, 1, 0, 1]
Output: 3
	import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,1,1,1,0,0,0,1,1,1,0};
      int max = 0;
      int counter = 0;
      for (int i = 0; i<arr.length; i++){
          if (arr[i] == 0){
            counter = 0;
          }
          else{
            counter++;
          }
          max = Integer.max(max, counter);
      }
      System.out.println(max);
    }
}
