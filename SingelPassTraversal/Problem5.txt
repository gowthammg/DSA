An array contains n distinct integers from the range 0 to n, with exactly one number missing. Find the missing number.
Input:arr = [3, 0, 1]
Output: 2
	import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{0,1,2,3,4};
      int[] ret = new int[arr.length+1];
      for (int i = 0; i<arr.length; i++){
        ret[arr[i]]++;
      }
      int output = ret.length;
      for (int i = 0;i<ret.length; i++){
        if (ret[i] == 0){
          output = i;
          break;
        }
      }
      System.out.println(output);
    }
}
