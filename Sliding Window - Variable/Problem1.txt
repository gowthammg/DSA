Find the Largest array where the contiguous subarray sum is equal to k 
	Input = [1,1,1,1,3,4,2,2]
	Output = [1,1,1,1]

	Approach 1:
	import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,1,1,1,4,2,2};
      int currsum = 0;
      Map<Integer, Integer> map = new HashMap<>();
      map.put(0, -1);
      int start = -1;
      int end = -1;
      int k = 4;
      int maxlen = 0;
      for (int i =0 ;i <arr.length; i++){
        currsum += arr[i];
        if (map.get(currsum-k) != null && maxlen < i-map.get(currsum-k)){
          start = map.get(currsum-k);
          end = i;
          maxlen = i-map.get(currsum-k);
        }
        map.put(currsum, i);
        //System.out.println(map);
      }
      System.out.println(maxlen+" "+start+" "+end);
    }
}
Approach 2:
import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{2,2,4,-4};
      int currsum = 0;
      int start = -1;
      int end = -1;
      int k = 4;
      int maxlen = 0;
      int i = 0, j = 0;
      while (j < arr.length){
        currsum += arr[j];
        j++;
        while (currsum > k && i < j){
          currsum -= arr[i];
          i++;
        }
        if (currsum == k && maxlen < j-i){
          maxlen = j-i;
          start = i;
          end = j;
        }
        //System.out.println(i+" "+j+" "+maxlen+" "+currsum);
      }
      System.out.println(maxlen+" "+start+" "+end);
    }
}
