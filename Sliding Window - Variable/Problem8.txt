8. Given a binary array containing only 0s and 1s and an integer k, find the longest contiguous subarray of 1s you can obtain by flipping at most k zeros to ones.
Input: arr = [1, 1, 0, 0, 1, 1, 1, 0, 1], k = 2
Output: [1, 1, 1, 1, 1, 0, 1]
import java.util.*;


public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,1,0,0,0,1,1,0,0,1,0,1};
      int k = 2;
      int i = 0; int j = 0;
      int start = -1, end = -1;
      int max = 0;
      int count = 0;
      while (j < arr.length){
        if (arr[j] == 0){
          count++;
        }
        j++;
        while (count > k){
          if (0 == arr[i])
            count--;
          i++;
        }
        if (max < (j-i)){
          max = j-i;
          start = i;
          end = j;
        }
        //System.out.println(map +" "+max);
      }
      System.out.println(max);
    }
}
