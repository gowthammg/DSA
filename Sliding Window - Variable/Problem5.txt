5.    Given a string, find the longest contiguous substring in which every character appears only once.
Input: s = "abcabcbb"
Output: "abc"
import java.util.*;


public class Main {
    public static void main(String[] args) {
      String input = "abcabcbbxyzabcdefghijklmnopqrstuvwxyz";
      int len = 0;
      int i = 0;
      int j = 0;
      int inputlen = input.length();
      int max = 0;
      int start = -1;
      int end = -1;
      Map<Character, Integer> map = new HashMap<>();
      while (j < inputlen){
          char curr = input.charAt(j);
          map.put(curr, map.getOrDefault(curr, 0)+1);
          j++;
          while (map.get(curr) > 1){
            char remchar = input.charAt(i);
            map.put(remchar, map.get(remchar)-1);
            i++;
          }
          if (max < (j-i)){
            max = j-i;
            start = i;
            end = j;
          }
      }
      System.out.println(max+" "+input.substring(start, end));
    }
}
