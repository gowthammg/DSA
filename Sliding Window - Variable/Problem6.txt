6. Given a string and an integer k, find the longest contiguous substring containing at most k distinct characters.
Input: s = "eceba", k = 2
Output: "ece"
import java.util.*;


public class Main {
    public static void main(String[] args) {
      String input = "uuuuuuuuuuuuuuvcvvccv";
      int k = 1;
      int i = 0; int j = 0;
      int start = -1, end = -1;
      int max = 0;
      Map<Character, Integer> map = new HashMap<>();
      while (j < input.length()){
        char curr = input.charAt(j);
        map.put(curr, map.getOrDefault(curr, 0)+1);
        j++;
        while (map.keySet().size() > k){
          char ichar = input.charAt(i);
          if (map.get(ichar) == 1){
            map.remove(ichar);
          }
          else{
            map.put(ichar, map.get(ichar)-1);
          }
          i++;
        }
        if (max < (j-i)){
          max = j-i;
          start = i;
          end = j;
        }
        //System.out.println(map +" "+max);
      }
      System.out.println(input.substring(start, end));
    }
}
