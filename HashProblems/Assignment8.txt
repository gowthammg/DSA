Return the distinct elements that appear in both arrays. Each common element must appear only once in the output.
Input: nums1 = [1, 2, 2, 1], nums2 = [2, 2, 3]
Output:[2]
import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] nums1 = new int[]{1,2,2,1};
      int[] nums2 = new int[]{2,2,3};
      Set<Integer> set1 = new HashSet<>();
      Set<Integer> set2 = new HashSet<>();
      for (Integer i : nums1){
        set1.add(i);
      }
      for (Integer i : nums2){
        set2.add(i);
      }
      Set<Integer> list = new HashSet<>();
      for (Integer i : set1){
        if (set2.contains(i))
          list.add(i);
      }
      for (Integer i : set2){
        if (set1.contains(i))
          list.add(i);
      }
      System.out.println(list);
    }
}
