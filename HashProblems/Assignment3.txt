import java.util.*;
public class Main {
    public static String checkMatch(Map<String, Map<Character, Integer>> map, String str){
      Map<Character, Integer> curmap = new HashMap<>();
      for (int i =0; i<str.length(); i++)
        curmap.put(str.charAt(i), curmap.getOrDefault(str.charAt(i), 0)+1);

      for (String key : map.keySet()){
        if (map.get(key).equals(curmap))
          return key;
      }
      return "";
    }

    public static Map<Character, Integer> findcount(String str){
      Map<Character, Integer> map = new HashMap<>();
      for (int i = 0; i<str.length(); i++)
        map.put(str.charAt(i), map.getOrDefault(str.charAt(i), 0)+1);
      return map;
    }

    public static void main(String[] args) {
      String[] arr = new String[]{"eat", "ant", "tea", "ate", "nat", "bat", "tan", "ate"};
      Map<String, List<String>> map = new HashMap<>();
      Map<String, Map<Character, Integer>> freq = new HashMap<>();
      for (String str : arr){
        String matchedkey = checkMatch(freq, str);
        if (matchedkey == ""){
          List<String> list = new ArrayList<>();
          list.add(str);
          map.put(str, list);
          freq.put(str, findcount(str));
        }
        else{
          List<String> list = map.get(matchedkey);
          list.add(str);
          map.put(matchedkey, list);
        }
      }
      System.out.println(map);
    }
}
