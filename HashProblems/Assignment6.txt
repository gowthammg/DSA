Return the total number of contiguous subarrays whose sum equals k. The array may contain positive numbers, negative numbers, and zeros.
Input: arr = [1, 2, 1, 2], k = 3
Output:3
import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,2,1,2};
      int k = 3;
      Map<Integer, Integer> map = new HashMap<>();
      map.put(0, 1);
      int total = 0;
      int sum = 0;
      for (int i = 0; i<arr.length; i++){
        sum += arr[i];
        if (map.get(sum-k) != null){
          total++;
        }
        map.put(sum, i);
      }
      System.out.println(total);
    }
}
