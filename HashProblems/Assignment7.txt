Return the length of the longest sequence of consecutive integers, regardless of their order in the input.
Input: arr = [100, 4, 200, 1, 3, 2]
Output: 4
import java.util.*;


public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{100, 4, 200, 1, 3, 2, 5, 6, 7, 8, 101, 102, 103, 104, 105, 106, 107, 108, 109};
      int max   = 0;
      Set<Integer> set = new HashSet<>();
      for (int i = 0;i <arr.length; i++){
        set.add(arr[i]);
      }
      for (int i = 0; i<arr.length; i++){
        int curr = 0;
        int curnum = arr[i];
        if (!set.contains(curnum-1)){
          while (set.contains(curnum)){
            curr++;
            curnum+=1;
          }
        }
        max = Integer.max(max, curr);
      }
      System.out.println(max);
    }
}
