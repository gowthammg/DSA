Given an integer array, move all negative numbers to the beginning and all non-negative numbers to the end. Maintain the relative order of elements within both groups.
Input: arr = [1, -2, 3, -4, 5, -6]
Output: [-2, -4, -6, 1, 3, 5]
import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{-5,-6,-7,-8,-9,10,1,2,3,4,6};
      int slow = 0, fast = 0;
      while (fast < arr.length){
        if (arr[fast] > 0){
          fast++;
        }
        else{
          int temp = arr[slow];
          arr[slow] = arr[fast];
          arr[fast] = temp;
          slow++;
          fast++;
        }
        //System.out.println(slow+" "+fast);
      }
      for (int i =0 ;i<arr.length; i++)
        System.out.print(arr[i]+" ");
    }
}
