Reverse an array without creating another array.
Input: arr = [1, 2, 3, 4, 5]
Output: [5, 4, 3, 2, 1]
	import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,2,3,4,5,6,7};
      int left = 0, right = arr.length-1;
      while (left < right){
        int temp = arr[left];
        arr[left] = arr[right];
        arr[right] = temp;
        left++;
        right--;
      }
      for (int i = 0; i<arr.length; i++)
        System.out.print(arr[i]+" ");
  }
}
