Given an array and an integer k, remove all occurrences of k in place.
Input: arr = [3, 2, 2, 3, 4, 2, 5], k =2
Output: [3, 3, 4, 5]
	
import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{2,2,2,2,5,6,7};
      int k = 2;
      int slow = 0, fast = 0;
      while (fast < arr.length){
        if (arr[fast] == k){
          fast++;
        }
        else{
          int temp = arr[slow];
          arr[slow] = arr[fast];
          arr[fast] = temp;
          slow++;
          fast++;
        }
      }
      for (int i =0 ;i<arr.length; i++)
        System.out.print(arr[i]+" ");
    }
}
