10. Given an array containing only 0, 1, and 2, sort the array in ascending order in place without using a sorting function.
Input: arr = [2, 0, 2, 1, 1, 0]
Output: [0, 0, 1, 1, 2, 2]

import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{0,1,2,1,1,2,2,0,0,0,1,1};
      int zero = 0;
      int one  = 0;
      int two = arr.length-1;
      while (one <= two){
        if (arr[one] == 0){
          int temp = arr[zero];
          arr[zero++] = arr[one];
          arr[one++]= temp;
        }
        else if (arr[one] == 2){
          int temp = arr[one];
          arr[one] = arr[two];
          arr[two--] = 2;
        }
        else{
          one++;
        }
        // for (int i = 0; i<arr.length; i++)
        //   System.out.print(arr[i]+" ");
        // System.out.println();
      }
    }
}
