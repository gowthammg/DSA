. Find the Maximum no of vowels in a substring of length k 
Input : abcedeeaiou
Output : 4
import java.util.*;
public class Main {
    public static boolean checkVowel(char a){
      if (a == 'a' || a == 'e' || a == 'i' || a == 'o' || a == 'u')
        return true;
      return false;
    }
    public static void main(String[] args) {
      String str = "rrrrtttrrtrtrtrtawfffio";
      int max = 0;
      int k = 4;
      int curr = 0;
      for (int i = 0; i<str.length(); i++){
          if (i < k ){
            if (checkVowel(str.charAt(i))){
              curr++;
            }
          }
          else{
            if (checkVowel(str.charAt(i-k)))
              curr--;
            if (checkVowel(str.charAt(i)))
             curr++;
          }
          if (max < curr)
            max = curr;
      }
      System.out.println(max+" ");
    }
}
