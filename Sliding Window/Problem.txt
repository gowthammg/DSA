Find the maximum sum of any three consecutive elements

	Input arr = [2,1,5,1,3,2]
	Output : 9

import java.util.*;
public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{2,1,9,1,-4,2};
      int k = 3;
      int curr = 0;
      int max  = 0;
      for (int i = 0; i<arr.length; i++){
          if (i < k){
            curr += arr[i];
          }
          else{
            curr -= arr[i-k];
            curr += arr[i];
          }
          if (max < curr)
            max = curr;
      }
      System.out.println(max);
    }
}
