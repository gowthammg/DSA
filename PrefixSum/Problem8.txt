8. Given an array of integers, calculate the sum of the sums of all possible contiguous subarrays.
class Solution {
    public int subarraySum(int[] arr) {
        // code here
        int sum = 0;
        int n = arr.length;
        for (int i = 0; i<arr.length; i++){
            sum += ((n-i)+(i*(n-i)))*arr[i];
        }
        return sum;
    }
}
