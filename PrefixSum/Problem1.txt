1. Given an array of integers, return an array where each element is the sum of all elements from index 0 to the current index.
Input: arr = [1, 2, 3, 4]
Output: [1, 3, 6, 10]

import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,2,3,4};
      for (int i = 1; i<arr.length; i++){
        arr[i] += arr[i-1];
        System.out.print(arr[i]+" ");
      }

    }
}
