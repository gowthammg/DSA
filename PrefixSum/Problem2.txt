2. Given an array of integers, find the index where the sum of elements on the left is equal to the sum of elements on the right. Exclude the current element from both sums. Return -1 if no such index exists.
Input: arr = [1, 3, 5, 2, 2]
Output: 2


import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,3,5,2,2};
      for (int i = 1; i<arr.length; i++){
        arr[i] += arr[i-1];
      }
      int ret = -1;
      for (int i = 1; i<arr.length; i++){
        //System.out.println(arr[i-1]+" "+(arr[arr.length-1]-arr[i]));
        if (arr[arr.length-1]-arr[i] == arr[i-1]){
          ret = i;
        }
      }
      System.out.print(ret);
    }
}
