2. Given an array of integers, find the index where the sum of elements on the left is equal to the sum of elements on the right. Exclude the current element from both sums. Return -1 if no such index exists.
Input: arr = [1, 3, 5, 2, 2]
Output: 2


import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,3,5,2,2};
      for (int i = 1; i<arr.length; i++){
        arr[i] += arr[i-1];
      }
      int ret = -1;
      for (int i = 1; i<arr.length; i++){
        //System.out.println(arr[i-1]+" "+(arr[arr.length-1]-arr[i]));
        if (arr[arr.length-1]-arr[i] == arr[i-1]){
          ret = i;
        }
      }
      System.out.print(ret);
    }
}

3. Given an array of integers, return an array where each element represents the absolute difference between the sum of elements to its left and the sum of elements to its right.
Input: arr = [10, 4, 8, 3]
Output: [15, 1, 11, 22]


import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{10,4,8,3};
      for (int i = 1; i<arr.length; i++){
        arr[i] += arr[i-1];
      }
      int[] ret = new int[arr.length];
      for (int i = 0; i<arr.length; i++){
        int prev = (i==0)? 0 : arr[i-1];
        ret[i] = Math.abs(prev-(arr[arr.length-1]-arr[i]));
        System.out.print(ret[i]+" ");
      }
    }
}


5. A person starts at altitude 0. Given an array representing the change in altitude at each point, find the highest altitude reached.
Input: arr = [-5, 1, 5, 0, -7]
Output: 1

import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{5,1,5,0,-7};
      int max = 0;
      for (int i = 1; i<arr.length; i++){
        arr[i] += arr[i-1];
        if (arr[i] > max){
          max = arr[i];
        }
      }
      System.out.println(max);
    }
}

6. Given an array of integers, return an array where each element is the product of all elements except the element at that index. Do not use division.
Input: arr = [1, 2, 3, 4]
Output: [24, 12, 8, 6]


import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{1,2,3,4};
      int[] ret = new int[arr.length];
      int[] prep =new int[arr.length];
      int[] posp = new int[arr.length];
      prep[0] = arr[0];
      for (int i =1; i<arr.length; i++){
        prep[i] = prep[i-1] * arr[i];
        //System.out.println(prep[i]+" ");
      }
      posp[posp.length-1] = arr[posp.length-1];
      for (int i=posp.length-2; i>=0; i--){
        posp[i] = posp[i+1] * arr[i];
        //System.out.println(posp[i]+" ");
      }
      for (int i = 0; i<ret.length; i++){
        if (i == 0){
          ret[i] = posp[i+1];
        }
        else if (i == ret.length-1){
          ret[i] = prep[i-1];
        }
        else{
          ret[i] = prep[i-1]*posp[i+1];
        }
        System.out.println(ret[i]+" ");
      }
    }
}
