3. Given an array of integers, return an array where each element represents the absolute difference between the sum of elements to its left and the sum of elements to its right.
Input: arr = [10, 4, 8, 3]
Output: [15, 1, 11, 22]


import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{10,4,8,3};
      for (int i = 1; i<arr.length; i++){
        arr[i] += arr[i-1];
      }
      int[] ret = new int[arr.length];
      for (int i = 0; i<arr.length; i++){
        int prev = (i==0)? 0 : arr[i-1];
        ret[i] = Math.abs(prev-(arr[arr.length-1]-arr[i]));
        System.out.print(ret[i]+" ");
      }
    }
}
