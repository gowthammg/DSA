Given a rotated sorted array that may contain duplicate elements, return True if the target exists; otherwise, return False. Input: arr = [2, 5, 6, 0, 0, 1, 2], target = 0 Output: True

import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{3,4,5,6,8,1,2};
      int target = 9;
      int left = 0;
      int right = arr.length-1;
      boolean res = false;
      while (left <= right){
        int mid = left + (right-left)/2;
        //System.out.println(left+" "+right+" "+mid);
        if (arr[mid] == target){
          res = true;
          break;
        }
        if (arr[left] == arr[mid] && arr[mid] == arr[right]){
          left++;
          right--;
          continue;
        }
        // if left to mid is sorted 
        if (arr[left] <= arr[mid]){
          if (arr[left] <= target && target < arr[mid]){
            right = mid-1;
          }
          else{
            left = mid+1;
          }
        }
        else{
          if (arr[mid] < target && target <= arr[right]){
            left = mid+1;
          }
          else{
            right = mid-1;
          }
        }
      }
      System.out.println(" "+res);
    }
}
