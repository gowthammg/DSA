2.Given a sorted array rotated at an unknown pivot, return the number of rotations performed. Assume all elements are distinct. Input: arr = [4, 5, 6, 7, 0, 1, 2] Output: 4

import java.util.*;

public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{3,4,5,1,2};
      int left = 0;
      int right = arr.length-1;
      int res = -1;
      while (left <= right){
        int mid = left + (right-left)/2;
        if (arr[left] <= arr[right]){
          res = left;
          break;
        }
        if (arr[mid] > arr[right]){
          left = mid+1;
        }
        else{
          right = mid;
        }

      }
      System.out.println(" "+res);
    }
}
