2.Given a sorted array rotated at an unknown pivot, return the number of rotations performed. Assume all elements are distinct. Input: arr = [4, 5, 6, 7, 0, 1, 2] Output: 4

import java.util.*;


public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{4,5,6,7,8,9,10,0,1,3};
      int left = 0;
      int right = arr.length-1;
      while (right-left > 1){
        int mid = left+(right-left)/2;
        //System.out.println(left+" "+right+" "+mid);
        if (arr[left] < arr[mid] && arr[mid] < arr[right]){
          right = mid-1;
        }
        else if (arr[mid] < arr[right]){
          right = mid;
        }
        else{
          left = mid;
        }


      }
      int ret = (arr[left] < arr[right])? left : right;
      System.out.println(ret);
    }
}
