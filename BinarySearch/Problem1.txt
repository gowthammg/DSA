Given a sorted array that has been rotated at an unknown pivot, return the minimum element. Assume all elements are distinct.
Input: arr = [4, 5, 6, 7, 0, 1, 2] Output: 0


import java.util.*;


public class Main {
    public static void main(String[] args) {
      int[] arr = new int[]{5,6,7,8,9,10,1,2,3,4};
      int left = 0;
      int right = arr.length-1;
      while (right-left > 1){
          int mid = left+(right-left)/2;
          //System.out.println(left+" "+right+" "+mid);
          if (arr[mid] >= arr[left] && arr[mid] <= arr[right]){
            right = mid-1;
          }
          else if (arr[mid] < arr[left]){
            right = mid;
          }
          else{
            left = mid;
          }
      }
      //System.out.println(left+" "+right);
      int ret = (arr[left] < arr[right])? left : right;
      System.out.println(ret);
    }
}
